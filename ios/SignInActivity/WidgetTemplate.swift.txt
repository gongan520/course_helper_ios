// MARK: - 内嵌插画素材
//
// 角色取自动画片原画面，按「迈步姿态」生成 8 帧后 base64 内嵌。
// Widget Extension 加图片资源要在 Xcode 工程里手工配 build phase，容易漏打包。
//
// 三套帧：
// - scene：小新牵小白整幅，用于展开态与锁屏（那里宽度足够，才走得出位移）
// - shin / dog：单人帧，用于紧凑态。胶囊两侧各只有十几 pt，整幅塞进去每个角色
//   只剩几像素，腿的摆动完全看不出来，所以紧凑态拆成单人各自迈步

@available(iOS 16.1, *)
enum SignInArt {
    static let sceneFrames: [UIImage?] = SceneFramesBase64.list.map { decode($0) }
    static let shinFrames: [UIImage?] = ShinFramesBase64.list.map { decode($0) }
    static let dogFrames: [UIImage?] = DogFramesBase64.list.map { decode($0) }

    static let sceneAspect: CGFloat = 219.0 / 132.0
    static let shinAspect: CGFloat = 88.0 / 132.0
    static let dogAspect: CGFloat = 151.0 / 96.0

    /// 取第 index 帧；任一帧解码失败时退到第 0 帧，避免整块空白
    static func image(_ frames: [UIImage?], _ index: Int) -> UIImage? {
        let n = frames.count
        guard n > 0 else { return nil }
        let i = ((index % n) + n) % n
        return frames[i] ?? frames[0]
    }

    private static func decode(_ s: String) -> UIImage? {
        guard let data = Data(base64Encoded: s, options: .ignoreUnknownCharacters) else {
            return nil
        }
        return UIImage(data: data)
    }
}

// MARK: - 步态参数

/// 把主 App 推送的 frame 换算成「该显示哪一帧姿态」以及身体的起伏与摇摆。
///
/// Live Activity 视图不能自播放动画，只能靠主 App 定时推新的 contentState 驱动。
/// 走路的样子来自两件事：腿部姿态逐帧变化（脚抬起、迈出去）＋ 身体随步伐上下起伏。
@available(iOS 16.1, *)
struct Gait {
    /// 该显示的步态帧下标
    let index: Int
    /// 身体起伏（pt）
    let bob: CGFloat
    /// 身体左右轻晃（度）
    let sway: Double

    /// - Parameters:
    ///   - swayAmp: 摇摆幅度（度）。紧凑态角色只有 25pt 高，腿摆只占几像素，
    ///     必须靠「企鹅式左右摇摆 + 上下颠簸」把走路的感觉放大，否则看不出在走
    ///   - bobRatio: 起伏占角色高度的比例
    init(frame: Int, height: CGFloat, swayAmp: Double = 3.2, bobRatio: CGFloat = 0.10) {
        let n = 8
        let g = ((frame % n) + n) % n
        index = g
        // 8 帧走完一个完整步态循环，身体起伏两次（左右脚各一步）
        let step = Double(g) / Double(n) * 2 * Double.pi
        let rise = abs(sin(step * 2))
        bob = CGFloat(rise) * max(1.4, height * bobRatio)
        sway = sin(step * 2) * swayAmp
    }
}

/// 整幅场景在横向轨道上往返走动（展开态 / 锁屏用）。
@available(iOS 16.1, *)
struct TrackState {
    let dx: CGFloat
    let mirrored: Bool

    init(frame: Int, travel: CGFloat) {
        let cycle = 16
        let p = ((frame % cycle) + cycle) % cycle
        let f = Double(p) / Double(cycle)
        // 前半个周期往右走，后半个周期转身往左走
        let forward = f < 0.5
        let t = forward ? f * 2 : (1 - f) * 2
        dx = CGFloat((t - 0.5) * 2) * travel
        mirrored = !forward
    }
}

// MARK: - 视图

/// 单人原地迈步：紧凑态用。
///
/// 胶囊两侧只有十几 pt，没有横移空间，所以这里不做位移 ——
/// 靠腿部姿态逐帧切换 + 明显上下起伏来表达「在走」。
@available(iOS 16.1, *)
struct SingleWalkView: View {
    var frames: [UIImage?]
    var aspect: CGFloat
    var frame: Int
    var height: CGFloat

    private var gait: Gait {
        Gait(frame: frame, height: height, swayAmp: 6.5, bobRatio: 0.12)
    }

    var body: some View {
        content
            .frame(width: height * aspect + 2, height: height + 8)
    }

    private var content: some View {
        Group {
            if let img = SignInArt.image(frames, gait.index) {
                Image(uiImage: img)
                    .resizable()
                    .scaledToFit()
                    .frame(height: height)
                    .rotationEffect(.degrees(gait.sway), anchor: .bottom)
                    .offset(y: -gait.bob)
            } else {
                Color.clear.frame(height: height)
            }
        }
    }
}

/// 小新牵小白的整幅场景，在轨道上往返走动：展开态 / 锁屏用。
///
/// 容器宽度必须大于图片宽度（= 图宽 + 2 × 行程），否则 `.offset` 一移动就被裁。
@available(iOS 16.1, *)
struct WalkSceneView: View {
    var frame: Int
    var height: CGFloat
    var travel: CGFloat

    private var gait: Gait { Gait(frame: frame, height: height) }
    private var track: TrackState { TrackState(frame: frame, travel: travel) }
    private var imageWidth: CGFloat { height * SignInArt.sceneAspect }

    var body: some View {
        content
            .frame(width: imageWidth + travel * 2, height: height + 6)
    }

    private var content: some View {
        Group {
            if let img = SignInArt.image(SignInArt.sceneFrames, gait.index) {
                Image(uiImage: img)
                    .resizable()
                    .scaledToFit()
                    .frame(height: height)
                    .scaleEffect(x: track.mirrored ? -1 : 1, y: 1, anchor: .center)
                    .rotationEffect(.degrees(track.mirrored ? -gait.sway : gait.sway), anchor: .bottom)
                    .offset(x: track.dx, y: -gait.bob)
            } else {
                Color.clear.frame(height: height)
            }
        }
    }
}

// MARK: - 锁屏 / 展开卡片

@available(iOS 16.1, *)
struct PetLockScreenView: View {
    let state: SignInActivityAttributes.ContentState

    private var headline: String {
        if state.statusText.contains("成功") { return "刚刚帮你签到成功啦～" }
        if state.statusText.contains("手动") { return "有个签到要你亲自来一下" }
        return SignInActivityKeys.slogan
    }

    var body: some View {
        HStack(alignment: .center, spacing: 6) {
            WalkSceneView(frame: state.frame, height: 44, travel: 16)

            VStack(alignment: .leading, spacing: 4) {
                Text(SignInActivityKeys.slogan)
                    .font(.system(size: 13, weight: .bold))
                    .foregroundColor(.white)
                    .lineLimit(1)
                    .minimumScaleFactor(0.7)

                Text(headline)
                    .font(.system(size: 11))
                    .foregroundColor(Color.white.opacity(0.62))
                    .lineLimit(1)
                    .minimumScaleFactor(0.8)

                HStack(spacing: 4) {
                    pill("托管 \(state.autoCount)", tint: ShinColor.red)
                    pill("定时 \(state.scheduledCount)", tint: ShinColor.yellow)
                    pill("手动 \(state.manualCount)", tint: Color.white.opacity(0.45))
                    pill("AI \(state.aiCount)", tint: Color(red: 0.35, green: 0.78, blue: 0.98))
                }
            }
            .layoutPriority(1)

            Spacer(minLength: 0)
        }
        .padding(.horizontal, 12)
        .padding(.vertical, 10)
    }

    private func pill(_ text: String, tint: Color) -> some View {
        HStack(spacing: 3) {
            Circle().fill(tint).frame(width: 4, height: 4)
            Text(text)
                .font(.system(size: 10, weight: .medium))
                .foregroundColor(Color.white.opacity(0.9))
                .lineLimit(1)
                .fixedSize()
        }
        .padding(.horizontal, 5)
        .padding(.vertical, 2.5)
        .background(Capsule().fill(Color.white.opacity(0.13)))
    }
}

// MARK: - Widget

@available(iOS 16.1, *)
struct SignInActivityWidget: Widget {
    var body: some WidgetConfiguration {
        ActivityConfiguration(for: SignInActivityAttributes.self) { context in
            PetLockScreenView(state: context.state)
                .activityBackgroundTint(Color.black)
                .activitySystemActionForegroundColor(Color.white)
                .widgetURL(URL(string: SignInActivityKeys.openURL))

        } dynamicIsland: { context in
            island(frame: context.state.frame)
        }
    }

    /// 灵动岛内容。单独写成普通函数，避免 @ViewBuilder 里出现 `let` / `return`。
    ///
    /// 紧凑态：中间那段约 125pt 是硬件开孔（前摄与传感器），iOS 不允许在上面绘制，
    /// 所以只能左右各摆一个角色。两个都在原地迈步，整条胶囊都在动。
    private func island(frame: Int) -> DynamicIsland {
        DynamicIsland {
            // 展开态：宽度足够，整幅「小新牵小白」在整行上来回走
            DynamicIslandExpandedRegion(.leading) {
                WalkSceneView(frame: frame, height: 40, travel: 30)
            }

            DynamicIslandExpandedRegion(.trailing) {
                Image(systemName: "pawprint.fill")
                    .font(.system(size: 13))
                    .foregroundColor(ShinColor.yellow.opacity(0.85))
            }

            DynamicIslandExpandedRegion(.bottom) {
                HStack(spacing: 6) {
                    Text(SignInActivityKeys.slogan)
                        .font(.system(size: 12, weight: .medium))
                        .foregroundColor(Color.white.opacity(0.92))
                        .lineLimit(1)
                    Spacer(minLength: 0)
                    Text("轻点回到 App")
                        .font(.system(size: 10))
                        .foregroundColor(ShinColor.yellow.opacity(0.9))
                }
            }
        } compactLeading: {
            SingleWalkView(
                frames: SignInArt.shinFrames,
                aspect: SignInArt.shinAspect,
                frame: frame,
                height: 25
            )
        } compactTrailing: {
            SingleWalkView(
                frames: SignInArt.dogFrames,
                aspect: SignInArt.dogAspect,
                frame: frame,
                height: 14
            )
        } minimal: {
            SingleWalkView(
                frames: SignInArt.shinFrames,
                aspect: SignInArt.shinAspect,
                frame: frame,
                height: 17
            )
        }
        .widgetURL(URL(string: SignInActivityKeys.openURL))
        .keylineTint(ShinColor.yellow)
    }
}

/// Extension 侧兜底读取 App Group 数据（活动状态异常时用于降级展示）。
@available(iOS 16.1, *)
func readSharedState() -> SignInActivityAttributes.ContentState {
    let defaults = UserDefaults(suiteName: SignInActivityKeys.appGroup)
    return SignInActivityAttributes.ContentState(
        statusText: defaults?.string(forKey: "live_status_text") ?? SignInActivityKeys.slogan,
        signedAccounts: defaults?.string(forKey: "live_signed_accounts") ?? "",
        autoCount: defaults?.integer(forKey: "live_auto_count") ?? 0,
        scheduledCount: defaults?.integer(forKey: "live_scheduled_count") ?? 0,
        manualCount: defaults?.integer(forKey: "live_manual_count") ?? 0,
        aiCount: defaults?.integer(forKey: "live_ai_count") ?? 0,
        mood: defaults?.string(forKey: "live_mood") ?? "idle",
        signedCount: defaults?.integer(forKey: "live_signed_count") ?? 0,
        frame: defaults?.integer(forKey: "live_frame") ?? 0
    )
}
